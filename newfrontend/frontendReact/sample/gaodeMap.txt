    <input id="input" type="text" placeholder="请输入骑行终点（如：天安门）" />

    <script type="text/javascript">
        window._AMapSecurityConfig = {
            securityJsCode: '我的安全密钥', // 替换为自己的
        }
    </script>
    <script type="text/javascript" src="https://webapi.amap.com/maps?v=1.4.15&key=我的key"></script>

    <script type="text/javascript">
        window.onload = function() {
            // 1. 初始化地图
            var map = new AMap.Map('container', {
                zoom: 18,
                resizeEnable: true
            });

            // 2. 高精度定位（保留原有逻辑）
            var marker = new AMap.Marker({
                position: new AMap.LngLat(116.39, 39.9),
                offset: new AMap.Pixel(-10, -10),
                icon: '//vdata.amap.com/icons/b18/1/2.png',
                title: '北京'
            });
            map.add(marker);

            AMap.plugin('AMap.Geolocation', function() {
                var geolocation = new AMap.Geolocation({
                    enableHighAccuracy: true,
                    timeout: 15000,
                    zoomToAccuracy: true,
                    buttonPosition: 'RB',
                    buttonOffset: new AMap.Pixel(10, 20)
                });
                map.addControl(geolocation);
                geolocation.getCurrentPosition();

                AMap.event.addListener(geolocation, 'complete', function(data) {
                    console.log('✅ 高精度定位结果：', data);
                    alert(
                        '定位成功！\n' +
                        '详细地址：' + data.formattedAddress + '\n' +
                        '街道：' + data.addressComponent.street + '\n' +
                        '经纬度：' + data.position.lng + ',' + data.position.lat
                    );
                    map.setCenter(data.position);
                    map.setZoom(18);
                });

                AMap.event.addListener(geolocation, 'error', function(err) {
                    console.error('❌ 定位失败：', err);
                    alert('定位失败！原因：' + err.info + '\n请按F12查看控制台详情');
                });
            });

            // 3. 搜索插件 + 骑行路线规划
            AMap.plugin(['AMap.Autocomplete', 'AMap.PlaceSearch', 'AMap.Riding'], function () {
                // 3.1 初始化搜索插件
                var autoOptions = {
                    city: "北京",
                    input: "input"
                };
                var autocomplete = new AMap.Autocomplete(autoOptions);
                var placeSearch = new AMap.PlaceSearch({
                    city: '北京',
                    map: map,
                    panel: false
                });

                // 3.2 初始化骑行插件（全局可访问）
                var riding = new AMap.Riding({
                    policy: AMap.RidingPolicy.LEAST_TIME,
                    map: map // 绑定地图，用于渲染路线
                });

                // 3.3 存储定位的起点（默认值，定位后更新）
                var startLngLat = [116.379028, 39.865042]; // 初始起点
                // 监听定位成功事件，更新起点为当前定位
                AMap.event.addListener(AMap.Geolocation, 'complete', function(data) {
                    startLngLat = [data.position.lng, data.position.lat];
                });

                // 3.4 搜索联想选中后，规划骑行路线
                AMap.event.addListener(autocomplete, 'select', function (e) {
                    placeSearch.search(e.poi.name, function (status, result) {
                        if (status === 'complete' && result.info === 'OK' && result.poiList.pois.length > 0) {
                            var endLngLat = [
                                result.poiList.pois[0].location.lng,
                                result.poiList.pois[0].location.lat
                            ];
                            // 执行骑行路线规划
                            planRidingRoute(startLngLat, endLngLat);
                        }
                    });
                });

                // 3.5 点击按钮手动触发路线规划
                document.getElementById('routeBtn').addEventListener('click', function() {
                    var inputVal = document.getElementById('input').value;
                    if (!inputVal) {
                        alert('请输入骑行终点！');
                        return;
                    }
                    placeSearch.search(inputVal, function (status, result) {
                        if (status === 'complete' && result.info === 'OK' && result.poiList.pois.length > 0) {
                            var endLngLat = [
                                result.poiList.pois[0].location.lng,
                                result.poiList.pois[0].location.lat
                            ];
                            planRidingRoute(startLngLat, endLngLat);
                        }
                    });
                });

                // 3.6 骑行路线规划核心方法
                function planRidingRoute(start, end) {
                    // 清除之前的路线
                    riding.clear();
                    // 搜索骑行路线
                    riding.search(start, end, function (status, result) {
                        if (status === 'complete' && result.info === 'OK') {
                            console.log('✅ 骑行路线规划成功：', result);
                            // 路线会自动渲染到map上（因绑定了map: map）
                            // 定位到路线起点
                            map.setCenter(start);
                            map.setZoom(15);
                        } else {
                            console.error('❌ 骑行路线规划失败：', result);
                            alert('路线规划失败！原因：' + result.info);
                        }
                    });
                }
            });
        }
    </script>
